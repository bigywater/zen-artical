LeetCode 37. 解数独

给定一个部分填好的 9×9 数独，把它填完。数独的规则是每行、每列、每个 3×3 宫里数字 1 到 9 各出现一次。做法是回溯：找一个空格，试填一个合法的数字，递归地去填后面的空格，走不通就撤销、换下一个数字。朴素的回溯按固定顺序填空格，能做但很慢；关键的加速是每一步先挑「候选数字最少」的空格：只有一个候选的空格是被"逼"出来的，没有候选的空格能让这条路立刻失败。本文用真实的填写次数展示这个加速的效果，给出五种语言的实现，并把求解过程做成分步演示。

官方示例有 51 个空格：朴素回溯放了 4208 次，其中 4157 次是走错路之后又被撤销的；候选最少优先只放了 51 次，也就是没有任何一次撤销，每一步都只有一个候选。耗时是 14.0ms 对 0.1ms。只有 21 个已知数字的那道题，两种做法都需要大量回溯，朴素回溯放了 49558 次（168.2ms），候选最少优先放了 14344 次（32.7ms）：次数少了约 3.5 倍，耗时少了约 5 倍。耗时比次数的差距更大，一个可能的原因是每次填写的开销不同：朴素回溯每次都要扫描整行整列整宫检查合法性，位掩码只需要按位运算；我没有单独测量这两部分的开销。这些数字只在这三道题上测得，不同题目的差距可能很不一样。

回溯加位掩码：五种语言的实现

用三个长度为 9 的整数数组 rows、cols、boxes，每个整数的第 d 位表示数字 d 是否已经在这行、这列、这宫里出现。把所有空格放在一个列表里，empties[0..k) 是已经填好的，empties[k..) 是还没填的。每次递归，扫描剩下的空格，算出每个空格的候选掩码（三个掩码取并再取反，只保留第 1 到 9 位），数出候选个数，挑最少的那个换到位置 k；如果有个空格候选数是 0，这条路走不通，直接返回；否则依次尝试这个空格的每个候选：填入、更新三个掩码、递归、撤销。Python、Java、Go、JavaScript、C++ 都是真实编译运行的。

Python · 解法代码def solve(board): g = [list(r) for r in board] rows, cols, boxes = [0] * 9, [0] * 9, [0] * 9 # 每行、每列、每个宫已经用过的数字（位掩码，第 d 位表示数字 d） empties = [] for r in range(9): for c in range(9): if g[r][c] == ".": empties.append((r, c)) else: bit = 1 << int(g[r][c]) rows[r] |= bit cols[c] |= bit boxes[r // 3 * 3 + c // 3] |= bit def dfs(k): # empties[0:k] 已经填好，empties[k:] 还是空格 if k == len(empties): return True best, best_mask, best_cnt = k, 0, 10 for i in range(k, len(empties)): # 挑「候选数字最少」的空格先填：失败得早，分支少 r, c = empties[i] mask = ~(rows[r] | cols[c] | boxes[r // 3 * 3 + c // 3]) & 0x3FE cnt = bin(mask).count("1") if cnt < best_cnt: best, best_mask, best_cnt = i, mask, cnt if cnt <= 1: break if best_cnt == 0: # 有个空格一个候选都没有：这条路走不通 return False empties[k], empties[best] = empties[best], empties[k] r, c = empties[k] b = r // 3 * 3 + c // 3 m = best_mask while m: bit = m & -m m ^= bit g[r][c] = str(bit.bit_length() - 1) rows[r] |= bit cols[c] |= bit boxes[b] |= bit if dfs(k + 1): return True rows[r] ^= bit # 回溯：撤销这次填写 cols[c] ^= bit boxes[b] ^= bit g[r][c] = "." return False dfs(0) return ["".join(r) for r in g]

Java · 解法代码class Solution { private int[] rows = new int[9], cols = new int[9], boxes = new int[9]; // 已经用过的数字（位掩码，第 d 位表示数字 d） private int[][] empties; private int n; public void solveSudoku(char[][] board) { List<int[]> list = new ArrayList<>(); for (int r = 0; r < 9; r++) { for (int c = 0; c < 9; c++) { if (board[r][c] == '.') { list.add(new int[]{r, c}); } else { int bit = 1 << (board[r][c] - '0'); rows[r] |= bit; cols[c] |= bit; boxes[r / 3 * 3 + c / 3] |= bit; } } } empties = list.toArray(new int[0][]); n = empties.length; dfs(board, 0); } private boolean dfs(char[][] board, int k) { // empties[0..k) 已经填好，empties[k..) 还是空格 if (k == n) return true; int best = k, bestMask = 0, bestCnt = 10; for (int i = k; i < n; i++) { // 挑「候选数字最少」的空格先填：失败得早，分支少 int r = empties[i][0], c = empties[i][1]; int mask = ~(rows[r] | cols[c] | boxes[r / 3 * 3 + c / 3]) & 0x3FE; int cnt = Integer.bitCount(mask); if (cnt < bestCnt) { best = i; bestMask = mask; bestCnt = cnt; if (cnt <= 1) break; } } if (bestCnt == 0) return false; // 有个空格一个候选都没有：这条路走不通 int[] tmp = empties[k]; empties[k] = empties[best]; empties[best] = tmp; int r = empties[k][0], c = empties[k][1], b = r / 3 * 3 + c / 3; for (int m = bestMask; m != 0; m &= m - 1) { int bit = m & -m; board[r][c] = (char) ('0' + Integer.numberOfTrailingZeros(bit)); rows[r] |= bit; cols[c] |= bit; boxes[b] |= bit; if (dfs(board, k + 1)) return true; rows[r] ^= bit; cols[c] ^= bit; boxes[b] ^= bit; // 回溯：撤销这次填写 } board[r][c] = '.'; return false; } }

JavaScript · 解法代码function solveSudoku(board) { // board 是 9×9 的字符数组，原地修改 const rows = new Array(9).fill(0), cols = new Array(9).fill(0), boxes = new Array(9).fill(0); // 已用数字的位掩码 const empties = []; const popcount = (x) => { let n = 0; while (x) { x &= x - 1; n++; } return n; }; for (let r = 0; r < 9; r++) { for (let c = 0; c < 9; c++) { if (board[r][c] === ".") { empties.push([r, c]); } else { const bit = 1 << Number(board[r][c]); rows[r] |= bit; cols[c] |= bit; boxes[Math.floor(r / 3) * 3 + Math.floor(c / 3)] |= bit; } } } const dfs = (k) => { // empties[0..k) 已经填好，empties[k..) 还是空格 if (k === empties.length) return true; let best = k, bestMask = 0, bestCnt = 10; for (let i = k; i < empties.length; i++) { // 挑「候选数字最少」的空格先填：失败得早，分支少 const [r, c] = empties[i]; const mask = ~(rows[r] | cols[c] | boxes[Math.floor(r / 3) * 3 + Math.floor(c / 3)]) & 0x3FE; const cnt = popcount(mask); if (cnt < bestCnt) { best = i; bestMask = mask; bestCnt = cnt; if (cnt <= 1) break; } } if (bestCnt === 0) return false; // 有个空格一个候选都没有：这条路走不通 [empties[k], empties[best]] = [empties[best], empties[k]]; const [r, c] = empties[k]; const b = Math.floor(r / 3) * 3 + Math.floor(c / 3); for (let m = bestMask; m; m &= m - 1) { const bit = m & -m; board[r][c] = String(31 - Math.clz32(bit)); rows[r] |= bit; cols[c] |= bit; boxes[b] |= bit; if (dfs(k + 1)) return true; rows[r] ^= bit; cols[c] ^= bit; boxes[b] ^= bit; // 回溯：撤销这次填写 } board[r][c] = "."; return false; }; dfs(0); }

C++ · 解法代码#include <utility> #include <vector> using namespace std; class Solution { int rows[9] = {0}, cols[9] = {0}, boxes[9] = {0}; // 已经用过的数字（位掩码，第 d 位表示数字 d） vector<pair<int, int>> empties; bool dfs(vector<vector<char>>& board, int k) { // empties[0..k) 已经填好，empties[k..) 还是空格 if (k == (int)empties.size()) return true; int best = k, bestMask = 0, bestCnt = 10; for (int i = k; i < (int)empties.size(); i++) { // 挑「候选数字最少」的空格先填：失败得早，分支少 int r = empties[i].first, c = empties[i].second; int mask = ~(rows[r] | cols[c] | boxes[r / 3 * 3 + c / 3]) & 0x3FE; int cnt = __builtin_popcount(mask); if (cnt < bestCnt) { best = i; bestMask = mask; bestCnt = cnt; if (cnt <= 1) break; } } if (bestCnt == 0) return false; // 有个空格一个候选都没有：这条路走不通 swap(empties[k], empties[best]); int r = empties[k].first, c = empties[k].second, b = r / 3 * 3 + c / 3; for (int m = bestMask; m; m &= m - 1) { int bit = m & -m; board[r][c] = '0' + __builtin_ctz(bit); rows[r] |= bit; cols[c] |= bit; boxes[b] |= bit; if (dfs(board, k + 1)) return true; rows[r] ^= bit; cols[c] ^= bit; boxes[b] ^= bit; // 回溯：撤销这次填写 } board[r][c] = '.'; return false; } public: void solveSudoku(vector<vector<char>>& board) { for (int r = 0; r < 9; r++) { for (int c = 0; c < 9; c++) { if (board[r][c] == '.') { empties.push_back({r, c}); } else { int bit = 1 << (board[r][c] - '0'); rows[r] |= bit; cols[c] |= bit; boxes[r / 3 * 3 + c / 3] |= bit; } } } dfs(board, 0); } };
