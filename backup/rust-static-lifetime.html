Box::leak

实测 · 本机执行输出 
  源码:
      1 | use std::alloc::{GlobalAlloc, Layout, System};
      2 | use std::sync::atomic::{AtomicUsize, Ordering};
      3 | struct C; static AL: AtomicUsize = AtomicUsize::new(0); static DE: AtomicUsize = AtomicUsize::new(0);
      4 | unsafe impl GlobalAlloc for C {
      5 |     unsafe fn alloc(&self, l: Layout) -> *mut u8 { AL.fetch_add(1, Ordering::Relaxed); unsafe { System.alloc(l) } }
      6 |     unsafe fn dealloc(&self, p: *mut u8, l: Layout) { DE.fetch_add(1, Ordering::Relaxed); unsafe { System.dealloc(p, l) } }
      7 | }
      8 | #[global_allocator] static G: C = C;
      9 | fn main() {
     10 |     let (a0, d0) = (AL.load(Ordering::Relaxed), DE.load(Ordering::Relaxed));
     11 |     for i in 0..1000 { let b = Box::new([i as u8; 64]); std::hint::black_box(&b); }          // 释放
     12 |     let (a1, d1) = (AL.load(Ordering::Relaxed), DE.load(Ordering::Relaxed));
     13 |     for i in 0..1000 { let b: &'static mut [u8; 64] = Box::leak(Box::new([i as u8; 64])); std::hint::black_box(&b); }   // 泄漏
     14 |     let (a2, d2) = (AL.load(Ordering::Relaxed), DE.load(Ordering::Relaxed));
     15 |     println!("1000 个 Box 正常 drop:alloc +{},dealloc +{}", a1 - a0, d1 - d0);
     16 |     println!("1000 个 Box::leak     :alloc +{},dealloc +{}(从不释放)", a2 - a1, d2 - d1);
     17 | }
  编译:成功
  运行(退出码 0):
    1000 个 Box 正常 drop:alloc +1000,dealloc +1000
    1000 个 Box::leak     :alloc +1000,dealloc +0(从不释放)
