LeetCode 87. 扰乱字符串

把一个字符串按下面的规则"扰乱"：如果长度是 1，停止；否则在某个位置把它切成两个非空子串 x 和 y，可以选择交换它们（得到 y + x）或者不交换（x + y），然后对 x 和 y 递归地做同样的操作。给定两个长度相同的字符串 s1 和 s2，判断 s2 是否可以由 s1 扰乱得到。直接按定义递归，每个位置有"交换"和"不交换"两种选择，规模会随长度指数增长。有两个关键的优化：记忆化（同一对子串只判断一次）和剪枝（两个子串的字母组成不同，一定不可能）。本文用真实统计的调用次数展示这两个优化的效果，给出五种语言的实现，并把递归的判断过程做成分步演示。

记忆化递归：五种语言的实现

dfs(i, j, L) 表示 s1 从下标 i 起长 L 的子串，能否扰乱成 s2 从下标 j 起长 L 的子串。先查缓存；两个子串相等返回真；字母的多重集合不同返回假；否则枚举切分位置 k（1 到 L-1），尝试两种情形：不交换（dfs(i, j, k) 且 dfs(i+k, j+k, L-k)）和交换（dfs(i, j+L-k, k) 且 dfs(i+k, j, L-k)），任一成立就返回真。状态一共有 O(n³) 个（i、j、L），每个状态枚举 O(n) 个切分位置，所以是 O(n⁴)。Python、Java、Go、JavaScript、C++ 都是真实编译运行的；Java、Go、C++ 用三维数组做缓存，JavaScript 用 Map。

Python · 解法代码def solve(s1, s2): n = len(s1) if n != len(s2): return False memo = {} def dfs(i, j, L): # s1[i:i+L] 能否扰乱成 s2[j:j+L] key = (i, j, L) if key in memo: return memo[key] a, b = s1[i:i + L], s2[j:j + L] if a == b: res = True elif sorted(a) != sorted(b): # 字母的多重集合不同，一定不行：提前剪枝 res = False else: res = False for k in range(1, L): # 在第 k 个字符处切开 if (dfs(i, j, k) and dfs(i + k, j + k, L - k)) or \ (dfs(i, j + L - k, k) and dfs(i + k, j, L - k)): # 不交换 / 交换两段 res = True break memo[key] = res return res return dfs(0, 0, n)

Java · 解法代码class Solution { private String s1, s2; private byte[][][] memo; // memo[i][j][L]：0 还没算，1 是 True，2 是 False public boolean isScramble(String s1, String s2) { int n = s1.length(); if (n != s2.length()) return false; this.s1 = s1; this.s2 = s2; memo = new byte[n][n][n + 1]; return dfs(0, 0, n); } private boolean dfs(int i, int j, int len) { // s1[i, i+len) 能否扰乱成 s2[j, j+len) if (memo[i][j][len] != 0) return memo[i][j][len] == 1; boolean res = false; if (s1.regionMatches(i, s2, j, len)) { res = true; } else if (sameLetters(i, j, len)) { // 字母的多重集合不同，一定不行：提前剪枝 for (int k = 1; k < len && !res; k++) { // 在第 k 个字符处切开 if ((dfs(i, j, k) && dfs(i + k, j + k, len - k)) || (dfs(i, j + len - k, k) && dfs(i + k, j, len - k))) res = true; // 不交换 / 交换两段 } } memo[i][j][len] = (byte) (res ? 1 : 2); return res; } private boolean sameLetters(int i, int j, int len) { int[] cnt = new int[128]; for (int k = 0; k < len; k++) { cnt[s1.charAt(i + k)]++; cnt[s2.charAt(j + k)]--; } for (int c : cnt) if (c != 0) return false; return true; } }

Go · 解法代码func isScramble(s1 string, s2 string) bool { n := len(s1) if n != len(s2) { return false } memo := make([][][]int8, n) // memo[i][j][L]：0 还没算，1 是 True，2 是 False for i := range memo { memo[i] = make([][]int8, n) for j := range memo[i] { memo[i][j] = make([]int8, n+1) } } sameLetters := func(i, j, l int) bool { var cnt [128]int for k := 0; k < l; k++ { cnt[s1[i+k]]++ cnt[s2[j+k]]-- } for _, c := range cnt { if c != 0 { return false } } return true } var dfs func(i, j, l int) bool // s1[i:i+l] 能否扰乱成 s2[j:j+l] dfs = func(i, j, l int) bool { if memo[i][j][l] != 0 { return memo[i][j][l] == 1 } res := false if s1[i:i+l] == s2[j:j+l] { res = true } else if sameLetters(i, j, l) { // 字母的多重集合不同，一定不行：提前剪枝 for k := 1; k < l && !res; k++ { // 在第 k 个字符处切开 if (dfs(i, j, k) && dfs(i+k, j+k, l-k)) || (dfs(i, j+l-k, k) && dfs(i+k, j, l-k)) { // 不交换 / 交换两段 res = true } } } if res { memo[i][j][l] = 1 } else { memo[i][j][l] = 2 } return res } return dfs(0, 0, n) }

JavaScript · 解法代码function isScramble(s1, s2) { const n = s1.length; if (n !== s2.length) return false; const memo = new Map(); // 键 "i,j,L" → 结果 const sameLetters = (i, j, len) => { const cnt = new Array(128).fill(0); for (let k = 0; k < len; k++) { cnt[s1.charCodeAt(i + k)]++; cnt[s2.charCodeAt(j + k)]--; } return cnt.every((c) => c === 0); }; const dfs = (i, j, len) => { // s1[i, i+len) 能否扰乱成 s2[j, j+len) const key = i + "," + j + "," + len; if (memo.has(key)) return memo.get(key); let res = false; if (s1.substr(i, len) === s2.substr(j, len)) { res = true; } else if (sameLetters(i, j, len)) { // 字母的多重集合不同，一定不行：提前剪枝 for (let k = 1; k < len && !res; k++) { // 在第 k 个字符处切开 if ((dfs(i, j, k) && dfs(i + k, j + k, len - k)) || (dfs(i, j + len - k, k) && dfs(i + k, j, len - k))) res = true; // 不交换 / 交换两段 } } memo.set(key, res); return res; }; return dfs(0, 0, n); }

C++ · 解法代码#include <string> #include <vector> using namespace std; class Solution { string s1, s2; vector<vector<vector<signed char>>> memo; // memo[i][j][L]：0 还没算，1 是 True，2 是 False bool sameLetters(int i, int j, int len) { int cnt[128] = {0}; for (int k = 0; k < len; k++) { cnt[(int)s1[i + k]]++; cnt[(int)s2[j + k]]--; } for (int c : cnt) if (c != 0) return false; return true; } bool dfs(int i, int j, int len) { // s1[i, i+len) 能否扰乱成 s2[j, j+len) if (memo[i][j][len] != 0) return memo[i][j][len] == 1; bool res = false; if (s1.compare(i, len, s2, j, len) == 0) { res = true; } else if (sameLetters(i, j, len)) { // 字母的多重集合不同，一定不行：提前剪枝 for (int k = 1; k < len && !res; k++) { // 在第 k 个字符处切开 if ((dfs(i, j, k) && dfs(i + k, j + k, len - k)) || (dfs(i, j + len - k, k) && dfs(i + k, j, len - k))) res = true; // 不交换 / 交换两段 } } memo[i][j][len] = res ? 1 : 2; return res; } public: bool isScramble(string a, string b) { int n = a.size(); if (n != (int)b.size()) return false; s1 = a; s2 = b; memo.assign(n, vector<vector<signed char>>(n, vector<signed char>(n + 1, 0))); return dfs(0, 0, n); } };

下面的演示用官方示例。缩进表示递归的深度。第一次在第 1 个字符处切开：g | reat，不交换（g↔r）和交换（g↔t）都因为字母不同被剪枝；再在第 2 个字符处切开：gr | eat，不交换要判断 gr↔rg 和 eat↔eat——gr 和 rg 字母相同，再切开，交换后 g↔g、r↔r 都相同，成立；eat 与 eat 相同，成立。所以 great 可以扰乱成 rgeat。每一帧都是把上面的解法在这组输入上真实执行记录下来的，运行器还检查了演示的最终结果与 solve 的返回值一致。
