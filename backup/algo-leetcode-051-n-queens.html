LeetCode 51. N 皇后

把 n 个皇后放到 n × n 的棋盘上，使任意两个皇后都不在同一行、同一列、同一条对角线上，返回所有不同的摆法（每种摆法用 n 个字符串表示棋盘，Q 是皇后，. 是空位）。做法是回溯：一行放一个皇后，逐行往下放，每放一个之前检查它所在的列和两条对角线有没有被占。判断对角线有一个巧妙的编码：同一条"主对角线"（\ 方向）上的格子 r - c 相同，同一条"副对角线"（/ 方向）上的格子 r + c 相同。本文用真实数据展示漏掉一条对角线会多出多少"解"，实测回溯与枚举全部排列的耗时，给出五种语言的实现，并把每一步的放置和冲突做成分步演示。

回溯加冲突集合：五种语言的实现

用三个集合（或布尔数组）记录被占用的列、主对角线（r - c，Java、Go、JavaScript、C++ 里加上 n - 1 使下标非负）、副对角线（r + c）。dfs(r) 尝试在第 r 行的每一列放皇后：三处都没被占才放；放完递归到下一行；返回时撤销。放满 n 行就把 queens（每行皇后所在的列）翻译成 n 个字符串收集起来。Python、Java、Go、JavaScript、C++ 都是真实编译运行的。

Python · 解法代码def solve(n): res = [] cols, diag1, diag2 = set(), set(), set() # 已经被占用的列、主对角线（r-c 相同）、副对角线（r+c 相同） queens = [] # queens[r] = 第 r 行皇后所在的列 def dfs(r): if r == n: res.append(["." * c + "Q" + "." * (n - c - 1) for c in queens]) return for c in range(n): if c in cols or r - c in diag1 or r + c in diag2: continue # 同列或同对角线：会互相攻击 cols.add(c) diag1.add(r - c) diag2.add(r + c) queens.append(c) dfs(r + 1) queens.pop() # 回溯：撤销这一行的皇后 cols.remove(c) diag1.remove(r - c) diag2.remove(r + c) dfs(0) return res

Java · 解法代码class Solution { private int n; private boolean[] cols, diag1, diag2; // 已占用的列、主对角线（r-c+n-1）、副对角线（r+c） private int[] queens; // queens[r] = 第 r 行皇后所在的列 private List<List<String>> res; public List<List<String>> solveNQueens(int n) { this.n = n; cols = new boolean[n]; diag1 = new boolean[2 * n]; diag2 = new boolean[2 * n]; queens = new int[n]; res = new ArrayList<>(); dfs(0); return res; } private void dfs(int r) { if (r == n) { List<String> board = new ArrayList<>(); for (int c : queens) { char[] row = new char[n]; Arrays.fill(row, '.'); row[c] = 'Q'; board.add(new String(row)); } res.add(board); return; } for (int c = 0; c < n; c++) { if (cols[c] || diag1[r - c + n - 1] || diag2[r + c]) continue; // 同列或同对角线：会互相攻击 cols[c] = diag1[r - c + n - 1] = diag2[r + c] = true; queens[r] = c; dfs(r + 1); cols[c] = diag1[r - c + n - 1] = diag2[r + c] = false; // 回溯：撤销这一行的皇后 } } }

Go · 解法代码func solveNQueens(n int) [][]string { res := [][]string{} cols := make([]bool, n) diag1 := make([]bool, 2*n) // 主对角线：r-c+n-1 相同 diag2 := make([]bool, 2*n) // 副对角线：r+c 相同 queens := make([]int, n) // queens[r] = 第 r 行皇后所在的列 var dfs func(r int) dfs = func(r int) { if r == n { board := make([]string, n) for i, c := range queens { board[i] = strings.Repeat(".", c) + "Q" + strings.Repeat(".", n-c-1) } res = append(res, board) return } for c := 0; c < n; c++ { if cols[c] || diag1[r-c+n-1] || diag2[r+c] { // 同列或同对角线：会互相攻击 continue } cols[c], diag1[r-c+n-1], diag2[r+c] = true, true, true queens[r] = c dfs(r + 1) cols[c], diag1[r-c+n-1], diag2[r+c] = false, false, false // 回溯：撤销这一行的皇后 } } dfs(0) return res }

JavaScript · 解法代码function solveNQueens(n) { const res = []; const cols = new Array(n).fill(false); const diag1 = new Array(2 * n).fill(false); // 主对角线：r-c+n-1 相同 const diag2 = new Array(2 * n).fill(false); // 副对角线：r+c 相同 const queens = new Array(n).fill(0); // queens[r] = 第 r 行皇后所在的列 const dfs = (r) => { if (r === n) { res.push(queens.map((c) => ".".repeat(c) + "Q" + ".".repeat(n - c - 1))); return; } for (let c = 0; c < n; c++) { if (cols[c] || diag1[r - c + n - 1] || diag2[r + c]) continue; // 同列或同对角线：会互相攻击 cols[c] = diag1[r - c + n - 1] = diag2[r + c] = true; queens[r] = c; dfs(r + 1); cols[c] = diag1[r - c + n - 1] = diag2[r + c] = false; // 回溯：撤销这一行的皇后 } }; dfs(0); return res; }
