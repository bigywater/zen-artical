真实实测 · Java 
// Solution.java
public class Solution {
    public static void main(String[] args) {
        String s = "   -42";
        int i = 0;
        while (i < s.length() && Character.isWhitespace(s.charAt(i))) i++;
        int sign = (i < s.length() && s.charAt(i) == '-') ? -1 : 1;
        if (i < s.length() && (s.charAt(i) == '+' || s.charAt(i) == '-')) i++;
        long value = 0;
        while (i < s.length() && Character.isDigit(s.charAt(i))) {
            value = value * 10 + (s.charAt(i) - '0');
            i++;
        }
        value *= sign;
        value = Math.max(Integer.MIN_VALUE, Math.min(Integer.MAX_VALUE, value));
        System.out.println(value);
    }
}

真实实测 · Go 
// solution.go
package main

import (
	"fmt"
	"math"
	"unicode"
)

func main() {
	s := "   -42"
	i := 0
	for i < len(s) && unicode.IsSpace(rune(s[i])) {
		i++
	}
	sign := 1
	if i < len(s) && s[i] == '-' {
		sign = -1
	}
	if i < len(s) && (s[i] == '+' || s[i] == '-') {
		i++
	}
	value := int64(0)
	for i < len(s) && s[i] >= '0' && s[i] <= '9' {
		value = value*10 + int64(s[i]-'0')
		i++
	}
	value *= int64(sign)
	if value < math.MinInt32 {
		value = math.MinInt32
	}
	if value > math.MaxInt32 {
		value = math.MaxInt32
	}
	fmt.Println(value)
}

// solution.cpp
#include <iostream>
#include <string>
#include <climits>
#include <cctype>
using namespace std;

int main() {
    string s = "   -42";
    int i = 0;
    while (i < (int)s.size() && isspace((unsigned char)s[i])) i++;
    int sign = (i < (int)s.size() && s[i] == '-') ? -1 : 1;
    if (i < (int)s.size() && (s[i] == '+' || s[i] == '-')) i++;
    long long value = 0;
    while (i < (int)s.size() && isdigit((unsigned char)s[i])) {
        value = value * 10 + (s[i] - '0');
        i++;
    }
    value *= sign;
    if (value < INT_MIN) value = INT_MIN;
    if (value > INT_MAX) value = INT_MAX;
    cout << value << endl;
    return 0;
}
